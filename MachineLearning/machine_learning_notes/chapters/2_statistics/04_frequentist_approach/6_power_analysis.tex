\subsection{Purpose}
Power analysis aims to \tB{determine the required sample size $n$} to detect an effect
of a given magnitude with specified confidence.

\subsection{Error types and power}
\begin{center}
    \begin{tabular}{|c|c|c|}
    \hline
    & \textbf{$H_{0}$ is True} & \textbf{$H_{1}$ is True}\\
    \hline
    \textbf{Do not reject $H_{0}$} & \tG{Correct: $1-\alpha$} & \tR{Type II Error: $\beta$}\\
    \hline
    \textbf{Reject $H_{0}$} & \tR{Type I Error: $\alpha$} & \tG{Correct: $1-\beta$ (Power)}\\
    \hline
    \end{tabular}
\end{center}
\begin{itemize}
    \item \textbf{Type I error} ($\alpha$): $\mathbb{P}(\text{reject } H_0 \mid H_0 \text{ true})$ — false positive
    \item \textbf{Type II error} ($\beta$): $\mathbb{P}(\text{fail to reject } H_0 \mid H_1 \text{ true})$ — false negative
    \item \textbf{Power} ($1-\beta$): $\mathbb{P}(\text{reject } H_0 \mid H_1 \text{ true})$ — true positive rate
\end{itemize}

\subsection{Power calculation}
For a \textbf{one-sample $z$-test}: $H_0: \mu = \mu_0$ vs $H_1: \mu = \mu_1$ (one-sided, $\mu_1 > \mu_0$).\\[0.5em]
\textbf{Step 1: Test statistic and its distributions}
$$Z = \dfrac{\bar{X} - \mu_0}{\sigma/\sqrt{n}}$$
\begin{itemize}
    \item Under $H_0$: $Z \sim \mathcal{N}(0, 1)$
    \item Under $H_1$: $Z \sim \mathcal{N}\left(\dfrac{\mu_1 - \mu_0}{\sigma/\sqrt{n}}, 1\right) = \mathcal{N}(\delta, 1)$
        where $\delta = \dfrac{\mu_1 - \mu_0}{\sigma/\sqrt{n}} = d\sqrt{n}$ is the \textbf{non-centrality parameter}
\end{itemize}
\textbf{Step 2: Rejection region under $H_0$}\\
Reject $H_0$ if $Z > z_{1-\alpha}$ (critical value from standard normal).\\[0.5em]
\textbf{Step 3: $Power = \mathbb{P}(\text{reject }H_0 | H_1\text{ true})$}
\begin{align*}
\text{Power} &= \mathbb{P}(Z > z_{1-\alpha} \mid H_1) \\
&= P\left(\dfrac{\bar{X} - \mu_0}{\sigma/\sqrt{n}} > z_{1-\alpha} \mid \mu = \mu_1\right)
\end{align*}
Standardizing under $H_1$ (subtract $\delta$ to get standard normal):
\begin{align*}
\text{Power} &= P\left(Z - \delta > z_{1-\alpha} - \delta\right) \\
&= P\left(Z' > z_{1-\alpha} - \delta\right) \quad \text{where } Z' \sim \mathcal{N}(0,1)
\end{align*}
\textbf{Step 4: Express using CDF}\\
Let $\Phi(z) = \mathbb{P}(Z' \leq z)$ be the standard normal CDF:
\begin{align*}
\text{Power} &= 1 - \mathbb{P}(Z' \leq z_{1-\alpha} - \delta) \\
&= 1 - \Phi(z_{1-\alpha} - \delta) \\
&= \Phi(\delta - z_{1-\alpha}) \quad \text{(using symmetry: } 1-\Phi(x) = \Phi(-x)\text{)}
\end{align*}
Substituting $\delta = d\sqrt{n}$:
$$\boxed{\text{Power} = \Phi\left(d\sqrt{n} - z_{1-\alpha}\right)}$$
\textbf{For two-sided test} ($H_1: \mu \neq \mu_0$): use $z_{1-\alpha/2}$ instead of $z_{1-\alpha}$:
$$\text{Power} \approx \Phi\left(d\sqrt{n} - z_{1-\alpha/2}\right)$$

\subsection{Sample size determination}
Solving for $n$ given target power $(1-\beta)$, effect size $d$, and significance $\alpha$:
$$n = \left(\dfrac{z_{1-\alpha/2} + z_{1-\beta}}{d}\right)^2$$
where $d = \dfrac{|\mu_1 - \mu_0|}{\sigma}$ is the standardized effect size.\\[0.5em]
\textbf{For common tests}:
\begin{itemize}
    \item \textbf{Two-sample $t$-test} (per group), \uB{assuming equal variances} ($\sigma_1^2 = \sigma_2^2 = \sigma^2$):
        $$n = 2\left(\dfrac{z_{1-\alpha/2} + z_{1-\beta}}{d}\right)^2 \quad \text{where } d = \dfrac{|\mu_1 - \mu_2|}{\sigma}$$
    \item \textbf{Two-sample $t$-test} (per group), \uB{unequal variances} (Welch's $t$-test):
        $$n_1 = \left(1 + \dfrac{\sigma_1^2}{\sigma_2^2}\right)\left(\dfrac{z_{1-\alpha/2} + z_{1-\beta}}{d}\right)^2
        \quad \text{where } d = \dfrac{|\mu_1 - \mu_2|}{\sqrt{\sigma_1^2 + \sigma_2^2}}$$
        For equal sample sizes with unequal variances: $n = \left(\dfrac{\sigma_1^2 + \sigma_2^2}{(\mu_1 - \mu_2)^2}\right)(z_{1-\alpha/2} + z_{1-\beta})^2$
    \item \textbf{Paired $t$-test}: $n = \left(\dfrac{z_{1-\alpha/2} + z_{1-\beta}}{d}\right)^2$ where $d = \dfrac{\mu_d}{\sigma_d}$
    \item \textbf{Correlation test}: $n = \left(\dfrac{z_{1-\alpha/2} + z_{1-\beta}}{0.5\ln\frac{1+r}{1-r}}\right)^2 + 3$
\end{itemize}

\subsection{Conducting a power analysis}
\textbf{A priori} (before data collection): determine $n$ given $\alpha$, power, and expected $d$
$$n = f(\alpha, 1-\beta, d) \quad \Rightarrow \quad \text{plan experiment}$$
\textbf{Post hoc} (after data collection): compute achieved power given $n$, $\alpha$, and observed $d$
$$1-\beta = g(\alpha, n, d) \quad \Rightarrow \quad \text{interpret results}$$
\textbf{Sensitivity}: determine minimum detectable effect size given $n$, $\alpha$, and power
$$d = h(\alpha, 1-\beta, n) \quad \Rightarrow \quad \text{assess study limitations}$$

\subsection{Example: two-sample $t$-test}
Given: $\alpha = 0.05$, power $= 0.80$, expected Cohen's $d = 0.5$ (medium effect)
\begin{align*}
z_{1-\alpha/2} &= z_{0.975} = 1.96 \\
z_{1-\beta} &= z_{0.80} = 0.84 \\
n &= 2\left(\dfrac{1.96 + 0.84}{0.5}\right)^2 = 2 \times (5.6)^2 \approx 63 \text{ per group}
\end{align*}
Total sample size: $N = 2n \approx 126$ participants.
